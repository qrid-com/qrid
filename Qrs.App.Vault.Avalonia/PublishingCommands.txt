/* (use this) AOT - works, smaller file, still needs 3 native DLLs */
dotnet publish Qrs.App.Vault.Avalonia.csproj --self-contained -r win-x64 -f net10.0 -c release -p:CreatePackage=false -p:PublishSingleFile=true -p:PublishAot=true

/* WORKS - produces (SINGLE FILE) that runs OK, but slower on a startup b/c it extracts to tmp folder */
dotnet publish Qrs.App.Vault.Avalonia.csproj --self-contained -r win-x64 -f net10.0 -c release -p:CreatePackage=false -p:PublishSingleFile=true -p:IncludeNativeLibrariesForSelfExtract=true

/* WORKS - produces (SINGLE FILE + 3 DLLs) that runs OK, but still needs 3 unmanaged DLLs - libHarfBuzzSharp.dll libSkiaSharp.dll av_libglesv2.dll */
dotnet publish Qrs.App.Vault.Avalonia.csproj --self-contained -r win-x64 -f net10.0 -c release -p:CreatePackage=false -p:PublishSingleFile=true


/* DEBUG */
dotnet publish Qrs.App.Vault.Avalonia.csproj --self-contained -r win-x64 -f net10.0 -c debug -p:CreatePackage=false -p:PublishSingleFile=true

/* this one works - produces EXE that runs, but it is NOT a single file, it requires all other DLL files */
dotnet publish Qrs.App.Vault.Avalonia.csproj -f net10.0-windows10.0.19041.0 -c Release -p:RuntimeIdentifierOverride=win-x64 -p:WindowsPackageType=None -p:WindowsAppSDKSelfContained=true
